subtract_min <- function(data) {
  min_values <- apply(data, 2, min, na.rm = TRUE)  # Calculate the minimum value for each column
  data <- data - min_values  # Subtract the minimum value from each element in the column
  return(data)
}

# Identify numeric columns except columns 1 and 2
numeric_cols <- names(OD600)[sapply(OD600, is.numeric) & !(names(OD600) %in% c("TimeS", "Time"))]

# Subtract the lowest value from each column
OD600[, (numeric_cols) := lapply(.SD, function(x) {
  if(length(x) > 1) {
    x - min(x, na.rm = TRUE)
  } else {
    x
  }
}), .SDcols = numeric_cols]
